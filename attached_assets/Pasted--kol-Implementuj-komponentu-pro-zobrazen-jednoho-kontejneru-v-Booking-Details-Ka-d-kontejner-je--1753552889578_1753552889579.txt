Úkol: Implementuj komponentu pro zobrazení jednoho kontejneru v Booking Details.
Každý kontejner je vizuálně oddělený card blokem s border, rounded a bg-gray-50.
Součástí je hlavní řádek (checkbox + info) a pod ním poznámky Carrier + Medlog vedle sebe.
Každá skupina může upravit pouze svou poznámku.
Dlouhá poznámka (více než 80 znaků nebo obsahující \n) zobrazí šipku ↘️.

📦 React komponenta – ContainerBlock.jsx

Copy
Edit
import React from "react";

const ContainerBlock = ({
  container,
  userGroup, // "carrier" nebo "medlog"
  onNoteChange, // (containerId, group, newNote) => void
}) => {
  const isCarrier = userGroup === "carrier";
  const isMedlog = userGroup === "medlog";

  const isLong = (text) => text.length > 80 || text.includes("\n");

  return (
    <div className="border border-gray-200 bg-gray-50 rounded-lg px-4 py-3 mb-4 shadow-sm">
      {/* Hlavní řádek */}
      <div className="flex items-center justify-between text-sm font-medium">
        <div className="flex items-center gap-2">
          <input type="checkbox" />
          <span>{container.jobNumber}</span>
        </div>
        <div>{container.containerNumber}</div>
        <div>{container.sizeType}</div>
        <div>{container.dateTime}</div>
        <div>{container.status}</div>
        <div>{container.address}</div>
      </div>

      {/* Poznámky */}
      <div className="flex gap-4 mt-3">
        {/* Carrier Note */}
        <div className="flex flex-col w-1/2">
          <label className="text-sm font-semibold">Carrier Note</label>
          <div className="relative">
            <input
              type="text"
              className={`w-full px-2 py-1 border rounded text-sm ${
                isCarrier ? "" : "bg-gray-100 text-gray-600"
              }`}
              readOnly={!isCarrier}
              value={container.carrierNote}
              onChange={(e) =>
                onNoteChange(container.id, "carrier", e.target.value)
              }
              placeholder="Enter carrier note"
            />
            {isLong(container.carrierNote) && (
              <span
                className="absolute right-1 top-1 text-red-500 cursor-pointer"
                title="Note is longer than one line"
                onClick={() => alert(container.carrierNote)}
              >
                ↘️
              </span>
            )}
          </div>
        </div>

        {/* Medlog Note */}
        <div className="flex flex-col w-1/2">
          <label className="text-sm font-semibold">Medlog Note</label>
          <div className="relative">
            <input
              type="text"
              className={`w-full px-2 py-1 border rounded text-sm ${
                isMedlog ? "" : "bg-gray-100 text-gray-600"
              }`}
              readOnly={!isMedlog}
              value={container.medlogNote}
              onChange={(e) =>
                onNoteChange(container.id, "medlog", e.target.value)
              }
              placeholder="Enter Medlog note"
            />
            {isLong(container.medlogNote) && (
              <span
                className="absolute right-1 top-1 text-red-500 cursor-pointer"
                title="Note is longer than one line"
                onClick={() => alert(container.medlogNote)}
              >
                ↘️
              </span>
            )}
          </div>
        </div>
      </div>
    </div>
  );
};

export default ContainerBlock;
