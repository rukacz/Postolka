🟠 Orange = change in any other field

Tooltip on hover: "X changes made by other group"

Dot disappears after acknowledgment.

jsx
Copy
Edit
// ChangeIndicatorDot.jsx
const ChangeIndicatorDot = ({ count, type }) => {
  if (count === 0) return null;
  const color = type === 'time' ? 'bg-red-500' : 'bg-orange-400';
  return (
    <div className={`relative w-6 h-6 rounded-full ${color} flex items-center justify-center text-white text-xs`}>
      {count}
      <span className="absolute top-full mt-1 text-xs text-gray-600 whitespace-nowrap">Changes by other group</span>
    </div>
  );
};
